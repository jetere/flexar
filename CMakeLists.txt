cmake_minimum_required(VERSION 3.24)
project(Flexar VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(flexar_core src/Graph.cpp)
target_include_directories(flexar_core PUBLIC src)
enable_testing()
add_executable(graph_tests tests/GraphTests.cpp)
target_link_libraries(graph_tests PRIVATE flexar_core)
add_test(NAME graph_tests COMMAND graph_tests)
if(WIN32)
  set(IMGUI "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui-1.91.8")
  if(NOT EXISTS "${IMGUI}/imgui.cpp")
    message(FATAL_ERROR "Run bootstrap.ps1 to download Dear ImGui 1.91.8 first.")
  endif()
  add_library(imgui STATIC ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_widgets.cpp ${IMGUI}/backends/imgui_impl_win32.cpp ${IMGUI}/backends/imgui_impl_dx12.cpp)
  target_include_directories(imgui PUBLIC ${IMGUI} ${IMGUI}/backends)
  add_library(flexar_runtime src/Renderer.cpp src/Media.cpp)
  target_compile_definitions(flexar_runtime PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(flexar_runtime PUBLIC flexar_core imgui d3d12 dxgi d3dcompiler dxguid windowscodecs mfplat mfreadwrite mfuuid ole32 comdlg32)
  add_executable(Flexar WIN32 src/Main.cpp)
  set_target_properties(Flexar PROPERTIES OUTPUT_NAME "flexar-canvas")
  target_link_libraries(Flexar PRIVATE flexar_runtime winmm)
  add_executable(media_tests tests/MediaTests.cpp)
  target_link_libraries(media_tests PRIVATE flexar_runtime)
  add_test(NAME media_tests COMMAND media_tests)
  add_test(NAME gpu_smoke COMMAND Flexar --smoke-test)
  if(MSVC)
    target_compile_options(Flexar PRIVATE /W4 /permissive- /utf-8)
    target_compile_options(flexar_core PRIVATE /W4 /permissive- /utf-8)
  endif()
  add_custom_target(flexar_assets ALL
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/shaders" "$<TARGET_FILE_DIR:Flexar>/shaders"
)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/assets/backgrounds")
    add_custom_command(TARGET flexar_assets POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets/backgrounds" "$<TARGET_FILE_DIR:Flexar>/assets/backgrounds")
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/assets/music/ambient.mp3")
    add_custom_command(TARGET flexar_assets POST_BUILD COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:Flexar>/assets/music" COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/assets/music/ambient.mp3" "$<TARGET_FILE_DIR:Flexar>/assets/music/ambient.mp3")
  endif()
  add_dependencies(Flexar flexar_assets)
endif()
